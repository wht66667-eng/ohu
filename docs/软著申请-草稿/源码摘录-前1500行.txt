class PageBoundary extends Component<{ children: ReactNode; onRetry: () => void }, { failed: boolean }> {
  state = { failed: false }
  static getDerivedStateFromError() { return { failed: true } }
  render() {
    if (!this.state.failed) return this.props.children
    return <section className="route-state" role="alert">
      <h1>页面暂时无法打开</h1>
      <p>请检查网络后重新加载。若仍打不开，可以刷新应用。</p>
      <div className="route-actions">
        <button type="button" onClick={this.props.onRetry}>重新加载</button>
        <button type="button" onClick={() => window.location.reload()}>刷新应用</button>
        <Link to={clientPath('/')}>返回首页</Link>
      </div>
    </section>
  }
}

function DeferredPage({ load }: { load: ScreenLoader }) {
  const [attempt, setAttempt] = useState(0)
  const Page = useMemo(() => lazy(load), [load, attempt])
  return <PageBoundary key={attempt} onRetry={() => setAttempt((value) => value + 1)}>
    <Suspense fallback={<RouteLoading />}><ReadyPage><Page /></ReadyPage></Suspense>
  </PageBoundary>
}

function ReadyPage({ children }: { children: ReactNode }) {
  useEffect(() => { void confirmOtaAppReady() }, [])
  return <><RouteEntryMotion />{children}</>
}

function ApplicationShell({ children }: { children: ReactNode }) {
  const location = useLocation()
  const isMeetRoute = location.pathname === '/meet' || location.pathname.startsWith('/meet/')
  const explicitStudent = new URLSearchParams(location.search).get('view') === 'student'
  let adminContext = false
  let studentPreview = explicitStudent
  try {
    studentPreview ||= sessionStorage.getItem('ohhu.web.student-preview') === '1'
    adminContext = sessionStorage.getItem('ohhu.web.admin-context') === '1'
  } catch { /* private browsing */ }
  useEffect(() => {
    try {
      if (explicitStudent) {
        sessionStorage.setItem('ohhu.web.student-preview', '1')
        sessionStorage.removeItem('ohhu.web.admin-context')
      } else if (isAdminWorkspacePath(location.pathname)) {
        sessionStorage.removeItem('ohhu.web.student-preview')
        sessionStorage.setItem('ohhu.web.admin-context', '1')
      }
    } catch { /* private browsing */ }
  }, [explicitStudent, location.pathname])
  // Existing shared screens keep their own routes and business logic. When
  // opened from this workspace, normalize their URL into its protected scope.
  // Invitation links are independently readable in a fresh mobile browser.
  // Joining a public activity still requires the normal account login.
  if (!IS_NATIVE && location.pathname.startsWith('/meet/')) return <PublicMeetShell>{children}</PublicMeetShell>
  if (!IS_NATIVE && !isMeetRoute && adminContext && !explicitStudent && !isAdminWorkspacePath(location.pathname)) {
    return <Navigate to={adminScopedPath(location.pathname, '/admin', false) + location.search + location.hash} replace />
  }
  if (!isMeetRoute && shouldOpenDesktopAdmin({ native: IS_NATIVE, pathname: location.pathname, width: window.innerWidth, studentPreview })) {
    return <Navigate to="/admin/content" replace />
  }
  if (!IS_NATIVE && isAdminWorkspacePath(location.pathname)) return <AdminWorkspace><AdminNotificationRuntime />{children}</AdminWorkspace>
  return <PhoneShell>{children}</PhoneShell>
}

/**
 * 按运行端选择路由模式：
 *  - H5：BrowserRouter，保留干净的路径 URL，配合托管侧 SPA 回退；
 *  - 套壳 App：HashRouter，`capacitor://` 协议下 pushState 深链与冷启动刷新会失效，
 *    hash 路由不依赖服务端回退，始终能落地。
 *
 * useLocation().pathname 在两种 Router 下都只返回 hash 之后的路径段，
 * 因此 TabBar / PhoneShell 里的字符串比较语义完全一致，页面无需改动。
 */
function AppRouter({ children }: { children: ReactNode }) {
  return IS_NATIVE ? <HashRouter>{children}</HashRouter> : <BrowserRouter>{children}</BrowserRouter>
}

  const shell = getAppShell()
  const home = shellHomePath(shell)
  if (shell !== 'user' && home !== '/') return <Navigate to={home} replace />
  return <ReadyPage><HomePage /></ReadyPage>
}

    </AuthProvider>
  )
}

/** 提醒弹窗里的到期时间：今天 20:30 / 明天 09:00 / MM-dd HH:mm */
function fmtTodoTime(iso: string) {
  const d = new Date(iso)
  if (Number.isNaN(d.getTime())) return ''
  const pad = (n: number) => String(n).padStart(2, '0')
  const hm = `${pad(d.getHours())}:${pad(d.getMinutes())}`
  const now = new Date()
  const sameDay = (a: Date, b: Date) =>
    a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth() && a.getDate() === b.getDate()
  if (sameDay(d, now)) return `今天 ${hm}`
  if (sameDay(d, new Date(now.getTime() + 86400000))) return `明天 ${hm}`
  return `${pad(d.getMonth() + 1)}-${pad(d.getDate())} ${hm}`
}

    /** 按条延后：id → 到期时间戳；兼容旧版 {id,until} / {ids,until} */
    type SnoozeMap = Record<string, number>

    function readSnoozeMap(): SnoozeMap {
      try {
        const raw = JSON.parse(localStorage.getItem(SNOOZE_KEY) || 'null')
        if (!raw || typeof raw !== 'object') return {}
        if (raw.map && typeof raw.map === 'object') return raw.map as SnoozeMap

      }

        if (Object.keys(cleaned).length) localStorage.setItem(SNOOZE_KEY, JSON.stringify({ map: cleaned }))
        else localStorage.removeItem(SNOOZE_KEY)
      } catch {
        /* ignore */
      }
      return cleaned
    }

    }

      try {
        const seen = JSON.parse(localStorage.getItem(LS_KEY) || '[]')
        return Array.isArray(seen) ? seen.map(String) : []
      } catch {
        return []
      }
    }

    function unmarkReminded(ids: number[]) {
      const dk = dayKey()
      const drop = new Set(ids.map((id) => `${id}@${dk}`))
      try {
        localStorage.setItem(LS_KEY, JSON.stringify(readSeen().filter((k) => !drop.has(k))))
      } catch {
        /* ignore */
      }
    }

    async function runScan(showTip: boolean) {
      const r = await todoRemindScan()
      if (stop || uidRef.current !== uid) return
      if (r.error || !r.data || !showTip) return

      let map = readSnoozeMap()
      const expiredIds = Object.entries(map)
        .filter(([, until]) => until <= Date.now())
        .map(([id]) => Number(id))
      if (expiredIds.length) {
        unmarkReminded(expiredIds)
        map = writeSnoozeMap(Object.fromEntries(Object.entries(map).filter(([, u]) => u > Date.now())))
      }

      const hide = new Set(
        Object.entries(map)
          .filter(([, until]) => until > Date.now())
          .map(([id]) => Number(id)),
      )
      const due = r.data.due.filter((d) => !hide.has(d.id))

      const wake = nextSnoozeWake(map)
      window.clearTimeout(snoozeTimer.current)
      if (wake > Date.now()) {
        snoozeTimer.current = window.setTimeout(() => void runScan(true).catch(() => undefined), wake - Date.now() + 200)
      }

      if (!due.length) return
      const seen = readSeen()
      const dk = dayKey()
      const fresh = due.filter((d) => !seen.includes(`${d.id}@${dk}`))
      if (fresh.length) setTodoTip({ ...r.data, due: fresh })
    }

    void runScan(true).catch(() => undefined)
    return () => {
      stop = true
      window.clearTimeout(snoozeTimer.current)
    }
  }, [auth.phase, auth.isSigned, uid])

  function markTipSeen() {
    if (!todoTip?.due.length) return

    const LS_KEY = `todo.reminded:${uid}`

    } catch {

    }
  }

  function dismissTip() {

  }

  function snoozeIds(ids: number[]) {
    if (!ids.length) return
    const until = Date.now() + 10 * 60 * 1000
    try {
      const raw = JSON.parse(localStorage.getItem(`todo.snooze:${uid}`) || 'null')
      const map: Record<string, number> =
        raw?.map && typeof raw.map === 'object' ? { ...raw.map } : {}
      for (const id of ids) map[String(id)] = until
      localStorage.setItem(`todo.snooze:${uid}`, JSON.stringify({ map }))
      const dayKey = localDayKey()
      const drop = new Set(ids.map((id) => `${id}@${dayKey}`))
      const seen = JSON.parse(localStorage.getItem(`todo.reminded:${uid}`) || '[]')
      if (Array.isArray(seen)) {
        localStorage.setItem(`todo.reminded:${uid}`, JSON.stringify(seen.filter((k) => !drop.has(String(k)))))
      }
    } catch {
      /* ignore */
    }
    window.clearTimeout(snoozeTimer.current)
    snoozeTimer.current = window.setTimeout(() => {
      void todoRemindScan().then((r) => {
        if (uidRef.current !== uid || r.error || !r.data?.due.length) return

        let seen: string[] = []
        try {
          seen = JSON.parse(localStorage.getItem(`todo.reminded:${uid}`) || '[]')
          if (!Array.isArray(seen)) seen = []
        } catch {
          seen = []
        }
        let map: Record<string, number> = {}
        try {
          const raw = JSON.parse(localStorage.getItem(`todo.snooze:${uid}`) || 'null')
          if (raw?.map) map = raw.map
        } catch {
          /* ignore */
        }

          Object.entries(map)
            .filter(([, u]) => Number(u) > Date.now())
            .map(([id]) => Number(id)),
        )
        const fresh = r.data.due.filter((d) => !hide.has(d.id) && !seen.includes(`${d.id}@${dayKey}`))
        if (fresh.length) setTodoTip({ ...r.data, due: fresh })
      }).catch(() => undefined)
    }, 10 * 60 * 1000 + 200)
  }

  function snoozeOne(id: number) {
    snoozeIds([id])
    setTodoTip((tip) => {
      if (!tip) return null
      const due = tip.due.filter((d) => d.id !== id)
      return due.length ? { ...tip, due } : null
    })
  }

  function snoozeTip() {
    const ids = todoTip?.due.map((d) => d.id) || []
    snoozeIds(ids)
    setTodoTip(null)
  }

        <div className={styles.serviceSave}><button type="button" className={styles.remindGhost} onClick={() => { setServiceDraft([...DEFAULT_HOME_SERVICE_IDS]); setServiceMessage(`已恢复${DEFAULT_HOME_SERVICE_IDS.map(id => HOME_SERVICE_OPTIONS.find(item => item.id === id)!.label).join('、')}，保存后生效。`) }}>恢复默认</button><button type="button" className={styles.remindMain} disabled={serviceDraft.length < 1 || serviceDraft.length > 6} onClick={saveServices}>保存调整</button></div>

          {([['light', '浅色'], ['dark', '深色'], ['system', '跟随系统']] as const).map(([value, label]) => <button key={value} type="button" aria-pressed={theme === value} onClick={() => { setTheme(value); applyTheme(value) }}>{label}</button>)}

      </DialogLayer> : null}

      {todoTip ? (
        <DialogLayer title="待办提醒" onClose={dismissTip}>
            <div className={styles.remindHead}>
              <span className={styles.remindEmoji} aria-hidden="true"><CampusIcon name="task" /></span>

              <button

                className={styles.remindMain}

                  markTipSeen()

                  navigate('/todo')

              >
                查看待办
              </button>
            </div>
        </DialogLayer>
      ) : null}
    </motion.div>
  )
}

function ServiceEntry({ item, kind = 'tile', artIndex, onUse }: {
  item: CampusService; kind?: EntryKind; artIndex?: number; onUse: (item: CampusService) => void
}) {
  const content = <>
    <span className={s.entryArt} aria-hidden="true">{kind === 'featured' && artIndex != null

      : <StudentServiceArtwork id={item.id} fallback={item.artwork} />}</span>
    <span className={s.entryCopy}><b>{item.title}</b>{kind === 'featured' || kind === 'result' ? <small>{item.desc}</small> : null}</span>
    {kind === 'result' && item.official ? <span className={s.official}>官方入口</span> : null}

    : <Link to={item.to} className={className} aria-label={label} onClick={() => onUse(item)}>{content}</Link>

/** 课名 → 稳定配色索引（同课同色，不随名称长度抖动） */
function colorOf(name: string) {
  let h = 0
  for (let i = 0; i < name.length; i++) h = (h * 31 + name.charCodeAt(i)) | 0
  return PALETTE[Math.abs(h) % PALETTE.length]
}

/** 节次区间 → 「08:05-09:45」（时间表可由用户自定义，走合成后的 tm） */
function sectionTime(tm: ReadonlyArray<readonly [string, string]>, a: number, b: number): string {
  const s = tm[clamp(a, 1, tm.length) - 1]
  const e = tm[clamp(b, 1, tm.length) - 1]
  return `${s[0]}-${e[1]}`
}

/** 把离散周次压缩成 1-3/5/8-12 形式 */
function compressWeeks(ws: number[]): string {
  const s = [...new Set(ws)].sort((a, b) => a - b)
  if (!s.length) return ''
  const parts: string[] = []
  let i = 0
  while (i < s.length) {
    let j = i
    while (j + 1 < s.length && s[j + 1] === s[j] + 1) j++
    parts.push(i === j ? `${s[i]}` : `${s[i]}-${s[j]}`)
    i = j + 1
  }
  return parts.join('/')
}

/** 今天是本周第几天（1=周一 … 7=周日） */
function todayDayIdx(): number {
  return ((new Date().getDay() + 6) % 7) + 1
}

/** 按 TERM_START 推算当前教学周 */
function guessCurrentWeek(): number {
  const start = new Date(`${TERM_START}T00:00:00`)
  const diff = Math.floor((Date.now() - start.getTime()) / (7 * 86400_000)) + 1
  return Math.max(1, diff)
}

/** 某教学周、某星期（1=周一…7=周日）对应的日历日期 */
function dateOfTeachingDay(week: number, day: number): Date {
  const start = new Date(`${TERM_START}T00:00:00`)
  const d = new Date(start)
  d.setDate(start.getDate() + (week - 1) * 7 + (day - 1))
  return d
}

function fmtMd(d: Date): string {
  return `${d.getMonth() + 1}/${d.getDate()}`
}

/** 当前处于第几节（用于时间轴高亮）；非上课时间返回 0 */
function currentSection(tm: ReadonlyArray<readonly [string, string]>): number {
  const mins = new Date().getHours() * 60 + new Date().getMinutes()
  for (let i = 0; i < tm.length; i++) {
    if (mins >= toMin(tm[i][0]) && mins <= toMin(tm[i][1])) return i + 1
  }
  return 0
}

function clamp(n: number, lo: number, hi: number) {
  return Math.min(hi, Math.max(lo, n))
}

/* ============================================================
 * 单元格分组：同天同课且节次连续/相邻的合并成一格
 * ============================================================ */

interface CellGroup {
  day: number
  name: string
  room: string
  teacher: string
  start: number
  end: number
  /** 这门课在本学期出现的所有周次 */
  allWeeks: number[]
}

function groupRows(rows: TtRow[], week: number): Record<number, CellGroup[]> {
  const filtered = rows.filter((r) => Number(r.week) === week)
  /* 先统计每门课（按 天|课名）在本学期出现过的所有周，供详情页展示 */
  const weeksOf = new Map<string, Set<number>>()
  for (const r of rows) {
    const k = `${r.day}|${r.name}`
    const set = weeksOf.get(k) ?? new Set<number>()
    set.add(Number(r.week))
    weeksOf.set(k, set)
  }
  const byKey: Record<string, Omit<CellGroup, 'start' | 'end'> & { ivs: Array<[number, number]> }> = {}
  for (const r of filtered) {
    const day = Number(r.day) || 1
    const ss = Array.isArray(r.sections) && r.sections.length ? r.sections : [1]
    const s0 = clamp(Math.min(...ss), 1, NS)
    const s1 = clamp(Math.max(...ss), s0, NS)
    const k = `${day}|${r.name}|${r.room || ''}|${r.teacher || ''}`
    const bucket = byKey[k] ?? (byKey[k] = { day, name: r.name, room: r.room, teacher: r.teacher, ivs: [], allWeeks: [] })
    bucket.ivs.push([s0, s1])
  }
  const out: Record<number, CellGroup[]> = {}
  for (const v of Object.values(byKey)) {
    v.ivs.sort((a, b) => a[0] - b[0])
    const merged: Array<[number, number]> = []
    for (const iv of v.ivs) {
      const last = merged[merged.length - 1]
      if (last && iv[0] <= last[1] + 1) last[1] = Math.max(last[1], iv[1])
      else merged.push([...iv])
    }
    const aw = [...(weeksOf.get(`${v.day}|${v.name}`) ?? [])]
    const list = merged.map<CellGroup>(([start, end]) => ({
      day: v.day,
      name: v.name,
      room: v.room,
      teacher: v.teacher,
      start,
      end,
      allWeeks: aw,
    }))
    out[v.day] = (out[v.day] ?? []).concat(list)
  }
  for (const k of Object.keys(out)) out[Number(k)].sort((a, b) => a.start - b.start)
  return out
}

}

  /* 成绩 / 考试已同步数量（教务缓存变化时刷新，供课表页入口展示） */
  const [academicCounts, setAcademicCounts] = useState(() => ({
    grades: loadGradesCache().length,
    exams: loadExamsCache().length,
  }))
  useEffect(() => {
    const onCache = () =>
      setAcademicCounts({ grades: loadGradesCache().length, exams: loadExamsCache().length })
    window.addEventListener(ACADEMIC_CACHE_EVENT, onCache)
    return () => window.removeEventListener(ACADEMIC_CACHE_EVENT, onCache)
  }, [])

  useEffect(() => {
    // 有缓存课表时刷新今日上课提醒
    if (loadScheduleCache().rows.length) {
      void rescheduleClassReminders().catch(() => undefined)
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [])

  // No school requests while browsing as a guest or before opening its form.
  useEffect(() => {
    if (!bindOpen || !auth.isSigned || !ownerSession.uid) return
    void loadCaptcha()
    return () => { captchaSequence.current++; captchaSession.current = null }
    // loadCaptcha uses the account-owned form and a request sequence.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [bindOpen, auth.isSigned])

  useEffect(() => {
    if (!resumedAction || resumedAction.id === handledAction.current || !auth.isSigned
      || !isQbankSessionCurrent(resumedAction.session)) return
    handledAction.current = resumedAction.id
    if (resumedAction.kind === 'bind') onReQuery()
    else if (resumedAction.kind === 'periods') openTmSheet()
    else if (resumedAction.kind === 'calendar') openCalSheet()
    else exportCalendar()
    // An explicit action survives the guest-to-signed form remount; no secrets do.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [resumedAction, auth.isSigned])

  /* 云端有保存过的节次时间则覆盖本地（换设备不丢） */
  useEffect(() => {
    const session = captureQbankSession()
    void (async () => {
      const remote = await fetchRemotePeriods()
      if (remote && isQbankSessionCurrent(session)) {
        setTm(remote)
        setTmCustom(isCustomPeriods())
      }
    })()
  }, [])

  function openTmSheet() {
    if (!requireSigned('periods')) return
    if (tmSaveBusy.current) return
    setTmDraft(tm.map((p) => [p[0], p[1]] as PeriodPair))
    setTmMsg('')
    setTmOpen(true)
  }

  function openCalSheet() {
    if (!requireSigned('calendar')) return
    setCalRules(listNearbyRules())
    setCalDate(toDateKey(new Date()))
    setCalKind('off')
    setCalTreatAs(1)
    setCalLabel('')
    setCalOpen(true)
  }

  function refreshCal() {
    setCalRules(listNearbyRules())
    setCalTick((n) => n + 1)
    void rescheduleClassReminders().catch(() => undefined)
  }

  function setDraftAt(i: number, j: 0 | 1, v: string) {
    if (tmSaveBusy.current) return
    setTmDraft((d) => d.map((p, k) => (k === i ? ((j === 0 ? [v, p[1]] : [p[0], v]) as PeriodPair) : p)))
  }

  async function saveTm() {
    if (!requireSigned('periods')) return
    if (tmSaveBusy.current) return
    const ok = sanitizePeriods(tmDraft)
    if (!ok) {
      setTmMsg('每节都要是合法的起止时间，且开始要早于结束')
      return
    }
    const session = captureQbankSession()
    tmSaveBusy.current = true
    setTmSaving(true)
    try {
      const r = await savePeriods(ok)
      if (!isQbankSessionCurrent(session)) return
      setTm(ok)
      setTmCustom(true)
      void rescheduleClassReminders().catch(() => undefined)
      if (r.error) {
        setTmMsg(`保存到云端失败：${r.error}（本机已生效）`)
        return
      }
      setTmOpen(false)
    } finally {
      tmSaveBusy.current = false
      if (isQbankSessionCurrent(session)) setTmSaving(false)
    }
  }

  function resetTmDraft() {
    if (tmSaveBusy.current) return
    const official = getOfficialPeriods()
    setTmDraft(official)
    setTmMsg(`已填入${officialSeasonLabel()}，点保存后生效`)
  }

  async function applyOfficialTm() {
    if (!requireSigned('periods')) return
    if (tmSaveBusy.current) return
    const session = captureQbankSession()
    const official = getOfficialPeriods()
    tmSaveBusy.current = true
    setTmSaving(true)
    try {
      const r = await savePeriods(official, { custom: false })
      if (!isQbankSessionCurrent(session)) return
      setTm(official)
      setTmDraft(official)
      setTmCustom(false)
      setTmMsg(r.error ? `已恢复官方（云端：${r.error}）` : `已恢复${officialSeasonLabel()}`)
      void rescheduleClassReminders().catch(() => undefined)
    } finally {
      tmSaveBusy.current = false
      if (isQbankSessionCurrent(session)) setTmSaving(false)
    }
  }

  function addCalRule() {
    if (!requireSigned('calendar')) return
    if (!/^\d{4}-\d{2}-\d{2}$/.test(calDate)) return
    const payload: Omit<CalendarOverride, 'source'> = {
      date: calDate,
      kind: calKind,
      label: calLabel.trim() || (calKind === 'off' ? '自定义放假' : `调课 · 上${dayName(calTreatAs)}的课`),
    }
    if (calKind === 'makeup') payload.treatAsDay = calTreatAs
    upsertUserOverride(payload)
    refreshCal()
  }

  /* 冷却倒计时：每秒递减，归零后自动恢复可提交 */
  useEffect(() => {
    if (!cooldown) return
    const t = setTimeout(() => setCooldown((c) => c - 1), 1000)
    return () => clearTimeout(t)
  }, [cooldown])

  /* 仅驱动顶部「下一节课」倒计时刷新（30 秒一次），与业务无关 */
  useEffect(() => {
    const t = setInterval(() => setNowTick(Date.now()), 30_000)
    return () => clearInterval(t)
  }, [])

  async function loadCaptcha() {
    if (!auth.isSigned || !ownerSession.uid || !bindVisible.current || !isQbankSessionCurrent(ownerSession)) return
    const session = captureQbankSession()
    const request = ++captchaSequence.current
    captchaSession.current = null
    setMsg('')
    setMsgErr(false)
    setCaptcha('')
    setSid('')
    setCapImg('')
    setCapLoading(true)
    try {
      const d = await fetchCaptcha()
      if (!bindVisible.current || request !== captchaSequence.current || !isQbankSessionCurrent(session)) return
      if (!d.sid || !d.image) throw new Error('验证码暂不可用，请重新加载')
      captchaSession.current = session
      setSid(d.sid)
      setCapImg(d.image)
    } catch (e) {
      if (!bindVisible.current || request !== captchaSequence.current || !isQbankSessionCurrent(session)) return
      const err = e as TimetableError
      setMsg(err.message || '验证码加载失败，请重试')
      setMsgErr(true)
      /* 取验证码也被限流时同样进入冷却，避免连点 */
      if (err.code === '1004') setCooldown(clamp(Math.round(err.retryAfter || 60), 5, 180))
    } finally {
      if (request === captchaSequence.current && isQbankSessionCurrent(session)) setCapLoading(false)
    }
  }

  async function onSubmit() {
    if (!requireSigned('bind') || cooldown > 0 || submitting) return
    if (!stuid.trim() || !pwd || !captcha.trim()) {
      showMsg('请填写完整学号、密码和验证码', true)
      return
    }
    if (!sid || !capImg || !captchaSession.current || !isQbankSessionCurrent(captchaSession.current)) {
      showMsg('验证码需要重新加载，请使用当前账号的验证码后重试', true)
      void loadCaptcha()
      return
    }
    const session = captureQbankSession()
    setSyncRequest('manual')
    showMsg('')
    try {
      const data = await queryAcademic(
        sid,
        {
          username: stuid.trim(),
          password: pwd,
          captcha: captcha.trim(),
        },
        ['schedule', 'grades', 'exams'],
      )
      if (!isQbankSessionCurrent(session)) return
      const gradesReceived = !data.grades_error && Array.isArray(data.grades)
      const examsReceived = !data.exams_error && Array.isArray(data.exams)
      const gradesSaved = gradesReceived && saveGradesCache(data.grades!, stuid.trim(), session)
      const examsSaved = examsReceived && saveExamsCache(data.exams!, stuid.trim(), session)
      const scheduleRows = data.rows || []
      setRows(scheduleRows)
      setPwd(rememberPwd ? pwd : '')
      const cacheSaved = saveScheduleCache(scheduleRows, stuid.trim(), session)
      const credentialsSaved = saveTimetableCredentials({
        stuid: stuid.trim(),
        password: pwd,
        rememberStuid: remember,
        rememberPassword: rememberPwd,
      }, session)
      const rem = cacheSaved ? await rescheduleClassReminders().catch(() => null) : null
      if (!isQbankSessionCurrent(session)) return
      if (examsSaved) void rescheduleExamReminders().catch(() => null)
      const extras: string[] = []
      if (!credentialsSaved) extras.push('本机账号信息未保存，请检查存储空间后重试')
      if (gradesSaved) extras.push(`成绩 ${data.grades!.length} 门`)
      else extras.push(data.grades_error ? '成绩未取到，已保留当前账号缓存' : '成绩保存失败，请检查存储空间')
      if (examsSaved) extras.push(`考试 ${data.exams!.length} 场`)
      else extras.push(data.exams_error ? '考试未取到，已保留当前账号缓存' : '考试保存失败，请检查存储空间')
      /*
       * 提醒要说清楚，不能静默：访客没有提醒归属（提醒按账号隔离，
       * personal-reminders 对空 uid 直接返回「等待登录状态确认」），
       * 而登录用户排不上时也要给出原因（课表为空 / 开关关着 / 通知权限没给）。
       */
      const notice = [
        cacheSaved
          ? scheduleRows.length ? '课表已更新' : '教务确认本学期暂无课程，已清除旧课程与提醒'
          : '课程已读取，但本机课表保存失败；请检查存储空间后重试',
        cacheSaved
          ? rem?.count
            ? `已安排 ${rem.count} 条上课提醒`
            : auth.isSigned
              ? (rem?.message ? `上课提醒未安排：${rem.message}` : '')
              : '登录后才会安排上课提醒（提醒按账号隔离）'
          : '',
        extras.length ? extras.join(' · ') : '',
      ]
        .filter(Boolean)
        .join('；')
      showMsg(notice, !cacheSaved || !gradesSaved || !examsSaved)
      const weeks = [...new Set(scheduleRows.map((r) => Number(r.week)).filter(Boolean))].sort(
        (a, b) => a - b,
      )
      if (weeks.length) setCurrentWeek(weeks.includes(thisWeek) ? thisWeek : weeks[0])
      return
    } catch (e) {
      if (!isQbankSessionCurrent(session)) return
      const err = e as TimetableError
      showMsg(err.message || '查询失败', true)
      if (err.code === '1001') {
        /* 验证码错/会话过期：自动换图并清空验证码输入（loadCaptcha 内已清空） */
        void loadCaptcha()
      } else if (err.code === '1004') {
        /* 被限流：按后端 Retry-After 做冷却倒计时，冷却期间禁用提交 */
        setCooldown(clamp(Math.round(err.retryAfter || 60), 5, 180))
      }
    } finally {
      if (isQbankSessionCurrent(session)) setSyncRequest(null)
    }
  }

  /**
   * 免手输：交给服务端 OCR 识别验证码（/api/timetable/auto）。
   * 成功后课表/成绩/考试一并落库并重排提醒（runAutoSync 内部已处理）；
   * 识别不出来时，把服务端**已经取好的**验证码图直接填进表单，用户接着手输即可——
   * 不会白跑一次请求，也不会和手输流程抢 sid。
   */
  async function onAutoSubmit() {
    if (!requireSigned('bind') || cooldown > 0 || submitting) return
    const id = stuid.trim()
